function sleeveDataVencimento(fabricacaoStr, meses){
  const [y,m,d] = fabricacaoStr.split('-').map(Number);
  return new Date(y, (m-1)+meses, d);
}
const SLEEVE_VALIDADE_URGENTE_DIAS = 30;
const SLEEVE_VALIDADE_ATENCAO_DIAS = 60;
function sleeveValidadeProgressoHTML(fabricacaoStr, meses){
  if(!meses) return '—';
  if(!fabricacaoStr) return `${meses} meses`;
  const fabricacao = new Date(...fabricacaoStr.split('-').map((v,i)=> i===1 ? Number(v)-1 : Number(v)));
  const vencimento = sleeveDataVencimento(fabricacaoStr, meses);
  vencimento.setHours(23,59,59,999);
  const agora = new Date();
  const totalDias = (vencimento - fabricacao) / 86400000;
  const diasRestantes = (vencimento - agora) / 86400000;
  const vencido = diasRestantes < 0;
  const diasInt = Math.ceil(Math.abs(diasRestantes));
  const pct = Math.max(6, Math.min(100, ((totalDias - diasRestantes) / totalDias) * 100));
  let classe, texto;
  if(vencido){
    classe = 'atrasado';
    texto = `Vencido há ${diasInt} dia${diasInt===1?'':'s'}`;
  } else if(diasRestantes <= SLEEVE_VALIDADE_URGENTE_DIAS){
    classe = 'urgente';
    texto = `Vence em ${diasInt} dia${diasInt===1?'':'s'}`;
  } else if(diasRestantes <= SLEEVE_VALIDADE_ATENCAO_DIAS){
    classe = 'atencao';
    texto = `Vence em ${diasInt} dia${diasInt===1?'':'s'}`;
  } else {
    classe = 'ok';
    texto = `Vence em ${diasInt} dia${diasInt===1?'':'s'}`;
  }
  return `
    <div class="entrega-progress-wrap">
      <div class="entrega-progress-label">${vencimento.toLocaleDateString('pt-BR')}</div>
      <div class="entrega-progress-bar"><div class="entrega-progress-fill ${classe}" data-pct="${pct.toFixed(1)}" style="width:0%"></div></div>
      <div class="entrega-progress-text ${classe}">${texto}</div>
    </div>`;
}

/* ========================================================================
   FLUXO DE SLEEVE
   Importa o relatório de fila de produção do ERP (rotina RGER339 — "Fila de
   Produção p/ Data - Pedido"; a mesma estrutura serve pra outras máquinas,
   ex: rotulagem). Mantém uma fila (sleeveOPs, upsert por OP — nunca apaga,
   pra dar tempo de validar OPs que já saíram da fila do ERP) e um histórico
   de recebimentos de matéria-prima por item (sleeveRecebimentos, em
   unidades) — o saldo em estoque é sempre a soma automática desses
   recebimentos.
   ======================================================================== */
function parseDataHoraBR(str){
  if(!str) return null;
  const [dataParte, horaParte] = String(str).trim().split(/\s+/);
  if(!dataParte) return null;
  const [d,m,y] = dataParte.split('/').map(Number);
  if(!d || !m || !y) return null;
  let hh = 0, mi = 0;
  if(horaParte){
    const [h,mn] = horaParte.split(':').map(Number);
    hh = h || 0; mi = mn || 0;
  }
  return new Date(y, m-1, d, hh, mi).toISOString();
}
function parseSleeveRGER339(texto){
  const ops = [];
  let maquinaAtual = null;
  texto.split(/\r?\n/).forEach(linha=>{
    if(!linha || !linha.trim()) return;
    const campos = linha.split(';');
    // linha de seção de máquina, ex: ";;;1902 - POSIMEC 02;;;..."
    const secao = (campos[3]||'').trim();
    if(/^\d+\s*-\s*.+/.test(secao)){
      maquinaAtual = secao.replace(/^\d+\s*-\s*/, '').trim();
      return;
    }
    // linha de dado: campo 0 é a OP (numérica) — descarta cabeçalhos/subtotais
    const opRaw = (campos[0]||'').trim();
    if(!/^\d+$/.test(opRaw)) return;

    const descricao = (campos[2]||'').trim();
    const mCodigo = descricao.match(/^(\d+\/\d+)-(.*)$/);
    const codigo = mCodigo ? fmtCodigo(mCodigo[1]) : '';
    const nome = mCodigo ? mCodigo[2].trim() : descricao;

    const versao = (campos[6]||'').trim();
    const mUnid = versao.match(/(\d+)\s*UNID/i);
    const unidPorCaixa = mUnid ? parseInt(mUnid[1],10) : null;
    const qtdCaixas = parseNumeroBR(campos[10]);

    ops.push({
      op: opRaw,
      codigo,
      nome,
      unidPorCaixa,
      qtdCaixas,
      qtdUnidades: unidPorCaixa!=null ? qtdCaixas*unidPorCaixa : null,
      pedido: (campos[7]||'').trim(),
      clienteNome: (campos[8]||'').trim(),
      termino: parseDataHoraBR(campos[16]),
      maquina: maquinaAtual || '—',
      validado: false,
      qtdValidada: null,
      saiuDaFila: false,
      saiuDaFilaEm: null,
    });
  });
  return ops;
}
// Confronto: uma OP some da fila quando o relatório reimportado não traz
// mais o número dela — nesse momento ela é marcada saiuDaFila (fica
// disponível pra validar em "Pedidos finalizados"), sem mexer em validado/
// qtdValidada (quem confirma a quantidade de verdade é o usuário). Se a OP
// reaparecer num import seguinte, a marcação é desfeita.
function mergeSleeveOPs(novasOPs){
  const opsNoRelatorio = new Set(novasOPs.map(o=>o.op));
  novasOPs.forEach(nova=>{
    const existente = sleeveOPs.find(o=>o.op===nova.op);
    if(existente){
      const {validado, qtdValidada} = existente;
      Object.assign(existente, nova, {validado, qtdValidada, saiuDaFila:false, saiuDaFilaEm:null});
    }else{
      sleeveOPs.push(nova);
    }
  });
  sleeveOPs.forEach(o=>{
    if(o.validado || opsNoRelatorio.has(o.op)) return;
    if(!o.saiuDaFila){
      o.saiuDaFila = true;
      o.saiuDaFilaEm = new Date().toISOString();
    }
  });
}
// Relatório RPCP264 ("Fila de Produção por Recurso") — não traz o número do
// pedido, só a OP e a data de Entrega. Como a OP já está ligada ao pedido
// certo (importada antes via RGER339), basta casar pelo número da OP pra
// anexar a data de entrega na OP já existente — não cria OP nova.
function parseSleeveRPCP264(texto){
  const entregas = [];
  texto.split(/\r?\n/).forEach(linha=>{
    if(!linha || !linha.trim()) return;
    const campos = linha.split(';');
    const opRaw = (campos[0]||'').trim();
    if(!/^\d+$/.test(opRaw)) return;
    const entrega = parseDataHoraBR(campos[16]);
    if(!entrega) return;
    entregas.push({ op:opRaw, entrega });
  });
  return entregas;
}
function mergeSleeveEntregas(entregas){
  let atualizados = 0, naoEncontrados = 0;
  entregas.forEach(({op, entrega})=>{
    const registro = sleeveOPs.find(o=>o.op===op);
    if(registro){ registro.entrega = entrega; atualizados++; }
    else naoEncontrados++;
  });
  return { atualizados, naoEncontrados };
}

function sleeveSemCoberturaCalc(){
  const coberturaPorCodigo = sleeveCoberturaPorCodigo();
  const alocacaoPorOP = sleeveAlocacaoPorCodigo(coberturaPorCodigo);
  const codigosComRecebimento = new Set(sleeveRecebimentos.map(r=>r.codigo));
  const pendentes = sleeveOPs.filter(o=>!o.validado && !o.saiuDaFila && codigosComRecebimento.has(o.codigo));
  const semCobertura = pendentes
    .filter(o=> alocacaoPorOP[o.op] && !alocacaoPorOP[o.op].cobre)
    .sort((a,b)=> new Date(a.entrega||a.termino||0) - new Date(b.entrega||b.termino||0));
  return { coberturaPorCodigo, alocacaoPorOP, pendentes, semCobertura };
}

  sleeveItensManuais.forEach(it=>{
    porCodigo[it.codigo] = { nome:it.nome, clienteNome:'—', pendenteCaixas:0, pendenteUnidades:0, ops:[] };
  });
  sleeveOPs.forEach(o=>{
    if(!o.codigo) return;
    if(!porCodigo[o.codigo]) porCodigo[o.codigo] = { nome:o.nome, clienteNome:o.clienteNome, pendenteCaixas:0, pendenteUnidades:0, ops:[] };
    porCodigo[o.codigo].nome = o.nome;
    porCodigo[o.codigo].clienteNome = o.clienteNome;
    if(!o.validado && !o.saiuDaFila){
      porCodigo[o.codigo].pendenteCaixas += (o.qtdCaixas||0);
      if(o.qtdUnidades!=null) porCodigo[o.codigo].pendenteUnidades += o.qtdUnidades;
      if(o.op) porCodigo[o.codigo].ops.push(o.op);
    }
  });
  Object.keys(porCodigo).forEach(cod=>{
    const recebimentosDoItem = sleeveRecebimentos.filter(r=>r.codigo===cod);
    const temRecebimento = recebimentosDoItem.length > 0;
    const totalRecebido = recebimentosDoItem.reduce((s,r)=>s+(r.quantidade||0),0);
    const totalSegregado = sleeveSegregados.filter(s=>s.codigo===cod).reduce((s,r)=>s+(r.quantidade||0),0);
    const saldoUnidades = totalRecebido - totalSegregado;
    porCodigo[cod].temRecebimento = temRecebimento;
    porCodigo[cod].saldoUnidades = saldoUnidades;
    porCodigo[cod].cobre = temRecebimento && saldoUnidades >= porCodigo[cod].pendenteUnidades;
  });
  return porCodigo;
}
// Aloca o saldo de cada código entre as OPs pendentes que precisam dele,
// priorizando quem tem a data de entrega mais próxima (a demanda real é
// quantidade + prazo, não a máquina) — assim, quando o saldo só dá pra
// cobrir os primeiros pedidos da fila, só os que vêm depois aparecem sem
// estoque, em vez de todo mundo aparecer com a mesma falta agregada.
function sleeveAlocacaoPorCodigo(coberturaPorCodigo){
  const porOP = {}; // op -> { cobre, faltam }
  const opsPorCodigo = {};
  sleeveOPs.forEach(o=>{
    if(!o.codigo || o.validado || o.saiuDaFila) return;
    (opsPorCodigo[o.codigo] = opsPorCodigo[o.codigo] || []).push(o);
  });
  const dataOrdenacao = o=> o.entrega ? new Date(o.entrega) : (o.termino ? new Date(o.termino) : new Date('9999-12-31'));
  Object.keys(opsPorCodigo).forEach(cod=>{
    const ops = opsPorCodigo[cod].slice().sort((a,b)=> dataOrdenacao(a) - dataOrdenacao(b));
    let saldoRestante = (coberturaPorCodigo[cod] && coberturaPorCodigo[cod].saldoUnidades) || 0;
    ops.forEach(o=>{
      const necessario = o.qtdUnidades || 0;
      if(saldoRestante >= necessario){
        porOP[o.op] = { cobre:true, faltam:0 };
        saldoRestante -= necessario;
      }else{
        porOP[o.op] = { cobre:false, faltam: necessario - saldoRestante };
        saldoRestante = 0;
      }
    });
  });
  return porOP;
}
// Tela "Saldo de estoque": um item (código) por linha, com o saldo (em
// unidades — soma automática dos recebimentos registrados no card
// "Recebimento de sleeve") e o quanto ainda está pendente de produzir
// daquele código (qtdCaixas × unidPorCaixa). Mostra TODOS os itens já

      if(sleeveEstoqueClienteFiltro && it.clienteNome !== sleeveEstoqueClienteFiltro) return false;
      if(!termo) return true;
      return cod.toLowerCase().includes(termo) || it.nome.toLowerCase().includes(termo) || it.clienteNome.toLowerCase().includes(termo);
    })
    .sort((a,b)=> itensPorCodigo[a].clienteNome.localeCompare(itensPorCodigo[b].clienteNome));

      <td class="val" style="text-align:center;">${fmtMilhar0(it.pendenteCaixas)}</td>

      <td class="val" style="text-align:center;">${it.cobre ? fmtMilhar0(it.saldoUnidades-it.pendenteUnidades) : '—'}</td>
      <td style="text-align:center;">${!it.temRecebimento ? '<span style="color:var(--text-faint);">—</span>' : (it.cobre ? '<span class="tag tag-repeticao">✓ COBRE</span>' : `<span class="tag tag-alterada">FALTAM ${fmtMilhar0(it.pendenteUnidades-it.saldoUnidades)} UNID.</span>`)}</td>
    </tr>`;
  }).join('') : `<tr><td colspan="8" class="hist-empty">${termo ? 'Nenhum item encontrado pra essa busca.' : 'Nenhum item importado ainda.'}</td></tr>`;
  tbodyEstoque.querySelectorAll('.sleeve-item-link').forEach(btn=>{
    btn.addEventListener('click', ()=> abrirSleeveOPsModal(btn.dataset.codigo));
  });

// Tela "Pedidos": todo número de pedido já visto em algum relatório
// importado, um por linha, com o cliente — pra gerenciar os pedidos sem
// precisar abrir a fila de produção por máquina. Cada pedido é classificado
// conforme a cobertura de estoque dos seus itens ainda pendentes (mesma
// base de "Saldo de estoque"): completo (todos os itens cobrem), sem
// estoque (nenhum item tem qualquer recebimento) ou incompleto (o resto —
// mistura de itens cobertos, faltando ou sem recebimento). Pedido sem
// nenhum item mais pendente (já produzido/validado) fica sem status. Um
// número de pedido nunca visto antes (fora de sleevePedidosConhecidos) cai
// em "a validar" até o usuário confirmar — só então passa a contar no card
// da classificação real e um snapshot vai pro "Histórico de pedidos".
const SLEEVE_PEDIDO_STATUS_LABEL = { 'a-validar':'Pedidos a validar', completo:'Pedidos completos', incompleto:'Pedidos incompletos', 'sem-estoque':'Pedidos sem estoque' };
// Itens do pedido com a data de Entrega de cada um (vem do relatório
// RPCP264, anexada na OP — ver mergeSleeveEntregas). Usa as OPs ainda
// pendentes do pedido; se não sobrar nenhuma (tudo já validado), cai pras
// OPs que o pedido teve, só pra exibição.
function sleevePedidoItensInfo(ped, coberturaPorCodigo){
  const opsDoPedido = sleeveOPs.filter(o=>o.pedido===ped);
  const relevantes = opsDoPedido.filter(o=>!o.validado && !o.saiuDaFila);
  const base = relevantes.length ? relevantes : opsDoPedido;
  const porCodigo = {};
  base.forEach(o=>{
    if(!o.codigo) return;
    if(!porCodigo[o.codigo]) porCodigo[o.codigo] = null;
    if(o.entrega && !porCodigo[o.codigo]) porCodigo[o.codigo] = o.entrega;
  });
  return Object.keys(porCodigo).map(c=>{
    const info = coberturaPorCodigo[c] || {};
    return { codigo:c, nome:info.nome||'', pendenteUnidades:info.pendenteUnidades||0, saldoUnidades:info.saldoUnidades||0, temRecebimento:!!info.temRecebimento, cobre:!!info.cobre, entrega:porCodigo[c] };
  });
}
function validarSleevePedido(ped, p, coberturaPorCodigo){
  if(!sleevePedidosConhecidos.includes(ped)) sleevePedidosConhecidos.push(ped);
  sleevePedidosHistorico.push({
    pedido: ped,
    clienteNome: p.clienteNome,
    status: p.status,
    itens: sleevePedidoItensInfo(ped, coberturaPorCodigo),
    validadoEm: new Date().toISOString(),
  });
}
function renderSleevePedidos(){
  const coberturaPorCodigo = sleeveCoberturaPorCodigo();
  const pedidosPorNumero = {};
  sleeveOPs.forEach(o=>{
    if(!o.pedido) return;
    if(!pedidosPorNumero[o.pedido]) pedidosPorNumero[o.pedido] = { clienteNome: o.clienteNome, codigosPendentes: new Set(), codigosTodos: new Set(), maquinas: new Set(), datasEntrega: new Set() };
    pedidosPorNumero[o.pedido].clienteNome = o.clienteNome;
    if(o.codigo) pedidosPorNumero[o.pedido].codigosTodos.add(o.codigo);
    if(o.entrega) pedidosPorNumero[o.pedido].datasEntrega.add(o.entrega);
    if(!o.validado && !o.saiuDaFila){
      if(o.codigo) pedidosPorNumero[o.pedido].codigosPendentes.add(o.codigo);
      pedidosPorNumero[o.pedido].maquinas.add(o.maquina || '—');
    }
  });
  Object.values(pedidosPorNumero).forEach(p=>{
    // Só preenche a coluna quando todos os itens do pedido têm a mesma
    // Entrega; se divergir (ou não tiver data ainda), fica em branco — o
    // detalhe por item continua no modal (clique no pedido).
    const datas = [...p.datasEntrega];
    p.entregaDisplay = datas.length === 1 ? datas[0] : null;
  });
  Object.values(pedidosPorNumero).forEach(p=>{
    const cods = [...p.codigosPendentes];
    if(!cods.length){ p.status = null; p.statusFinal = null; return; }
    const todosCobrem = cods.every(c=> coberturaPorCodigo[c] && coberturaPorCodigo[c].cobre);
    const nenhumTemEstoque = cods.every(c=> !coberturaPorCodigo[c] || !coberturaPorCodigo[c].temRecebimento);
    p.status = todosCobrem ? 'completo' : (nenhumTemEstoque ? 'sem-estoque' : 'incompleto');
  });
  Object.entries(pedidosPorNumero).forEach(([ped,p])=>{
    p.statusFinal = (p.status && !sleevePedidosConhecidos.includes(ped)) ? 'a-validar' : p.status;
  });

  // Filtro combinado (status + máquina + cliente + busca). "ignorar" deixa
  // de fora a própria dimensão que está sendo contada, pra cada linha de
  // cards refletir só as OUTRAS restrições já ativas (faceted filtering) —
  // assim um card só aparece/soma se realmente sobrar pedido com aquele
  // corte, considerando o que já está filtrado nas outras linhas.
  const termoAtual = sleevePedidosSearchTerm.trim().toLowerCase();
  function pedidoPassaFiltros(ped, p, ignorar){
    ignorar = ignorar || {};
    if(!ignorar.status && filtroStatusPedidoSleeve && p.statusFinal !== filtroStatusPedidoSleeve) return false;
    if(!ignorar.maquina && filtroMaquinaPedidoSleeve && !p.maquinas.has(filtroMaquinaPedidoSleeve)) return false;
    if(sleevePedidosClienteFiltro && p.clienteNome !== sleevePedidosClienteFiltro) return false;
    if(termoAtual && !(ped.toLowerCase().includes(termoAtual) || p.clienteNome.toLowerCase().includes(termoAtual))) return false;
    return true;
  }

  const wrapStatus = document.getElementById('sleeve-pedidos-status-cards');
  const contagens = { 'a-validar':0, completo:0, incompleto:0, 'sem-estoque':0 };
  Object.entries(pedidosPorNumero).forEach(([ped,p])=>{
    if(p.statusFinal && pedidoPassaFiltros(ped, p, {status:true})) contagens[p.statusFinal]++;
  });
  if(filtroStatusPedidoSleeve && !contagens.hasOwnProperty(filtroStatusPedidoSleeve)) filtroStatusPedidoSleeve = null;
  wrapStatus.innerHTML = Object.keys(SLEEVE_PEDIDO_STATUS_LABEL).map(st=>{
    const isActive = filtroStatusPedidoSleeve === st;
    return `
      <div class="stage-mini-card ${isActive?'active':''}" data-status="${st}">
        <div class="smc-nome">${SLEEVE_PEDIDO_STATUS_LABEL[st]}</div>
        <div class="smc-count ${contagens[st]===0?'zero':''}">${contagens[st]}</div>
      </div>`;
  }).join('');
  wrapStatus.querySelectorAll('.stage-mini-card').forEach(card=>{
    card.addEventListener('click', ()=>{
      const st = card.dataset.status;
      filtroStatusPedidoSleeve = (filtroStatusPedidoSleeve === st) ? null : st;
      renderSleevePedidos();
    });
  });

  // Cards de máquina: só aparece a máquina que ainda tem pelo menos um
  // pedido depois de aplicar o status/cliente/busca já selecionados.
  const contagemMaquinas = {};
  Object.entries(pedidosPorNumero).forEach(([ped,p])=>{
    if(!pedidoPassaFiltros(ped, p, {maquina:true})) return;
    p.maquinas.forEach(m=>{ contagemMaquinas[m] = (contagemMaquinas[m]||0) + 1; });
  });
  const todasMaquinasPedido = Object.keys(contagemMaquinas).sort();
  if(filtroMaquinaPedidoSleeve && !contagemMaquinas[filtroMaquinaPedidoSleeve]) filtroMaquinaPedidoSleeve = null;
  const wrapMaquinaCards = document.getElementById('sleeve-pedidos-maquina-cards');
  wrapMaquinaCards.innerHTML = todasMaquinasPedido.map(maq=>{
    const isActive = filtroMaquinaPedidoSleeve === maq;
    return `
      <div class="stage-mini-card ${isActive?'active':''}" data-maquina="${maq}">
        <div class="smc-nome">${maq}</div>
        <div class="smc-count">${contagemMaquinas[maq]}</div>
      </div>`;
  }).join('');
  wrapMaquinaCards.querySelectorAll('.stage-mini-card').forEach(card=>{
    card.addEventListener('click', ()=>{
      const maq = card.dataset.maquina;
      filtroMaquinaPedidoSleeve = (filtroMaquinaPedidoSleeve === maq) ? null : maq;
      renderSleevePedidos();
    });
  });

  const selectCliente = document.getElementById('sleeve-pedidos-cliente-filter');
  const clientesUnicos = [...new Set(Object.values(pedidosPorNumero).map(p=>p.clienteNome))].sort((a,b)=>a.localeCompare(b));
  if(sleevePedidosClienteFiltro && !clientesUnicos.includes(sleevePedidosClienteFiltro)) sleevePedidosClienteFiltro = '';
  selectCliente.innerHTML = `<option value="">Todos os clientes</option>` + clientesUnicos.map(c=>`<option value="${c}">${c}</option>`).join('');
  selectCliente.value = sleevePedidosClienteFiltro;

  const tbody = document.querySelector('#sleeve-pedidos-table tbody');
  const pedidosLista = Object.keys(pedidosPorNumero)
    .filter(ped=> pedidoPassaFiltros(ped, pedidosPorNumero[ped]))
    .sort((a,b)=> pedidosPorNumero[a].clienteNome.localeCompare(pedidosPorNumero[b].clienteNome) || a.localeCompare(b));
  const SITUACAO_TAG = {
    completo: '<span class="tag tag-repeticao">✓ COMPLETO</span>',
    incompleto: '<span class="tag tag-confirmacao">INCOMPLETO</span>',
    'sem-estoque': '<span class="tag tag-alterada">SEM ESTOQUE</span>',
  };
  tbody.innerHTML = pedidosLista.length ? pedidosLista.map(ped=>{
    const p = pedidosPorNumero[ped];
    const situacao = p.statusFinal === 'a-validar'
      ? `<button class="btn small gold sleeve-pedido-validar-btn" data-pedido="${ped}">Validar</button>`
      : (SITUACAO_TAG[p.status] || '<span style="color:var(--text-faint);">—</span>');
    const entrega = p.entregaDisplay ? fmtDate(p.entregaDisplay) : '<span style="color:var(--text-faint);">—</span>';
    return `
    <tr>
      <td class="pedido"><button class="sleeve-item-link" data-pedido="${ped}" style="background:none; border:none; padding:0; text-align:left; cursor:pointer; color:inherit; font:inherit; text-decoration:underline;">${ped}</button></td>
      <td>${p.clienteNome}</td>
      <td style="text-align:center;">${situacao}</td>
      <td style="text-align:center;">${entrega}</td>
    </tr>`;
  }).join('') : `<tr><td colspan="4" class="hist-empty">${termoAtual || sleevePedidosClienteFiltro || filtroStatusPedidoSleeve || filtroMaquinaPedidoSleeve ? 'Nenhum pedido encontrado.' : 'Nenhum pedido importado ainda.'}</td></tr>`;

  tbody.querySelectorAll('.sleeve-item-link').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      const ped = btn.dataset.pedido;
      const p = pedidosPorNumero[ped];
      abrirSleevePedidoItensModal(ped, p.clienteNome, sleevePedidoItensInfo(ped, coberturaPorCodigo));
    });
  });
  tbody.querySelectorAll('.sleeve-pedido-validar-btn').forEach(btn=>{
    btn.addEventListener('click', async ()=>{
      const ped = btn.dataset.pedido;
      validarSleevePedido(ped, pedidosPorNumero[ped], coberturaPorCodigo);
      await saveState();
      renderTudoSleeve();
    });
  });

  const btnValidarTodos = document.getElementById('sleeve-pedidos-validar-todos-btn');
  const pendentesDeValidar = pedidosLista.filter(ped=> pedidosPorNumero[ped].statusFinal === 'a-validar');
  btnValidarTodos.textContent = pendentesDeValidar.length ? `Validar todos (${pendentesDeValidar.length})` : 'Validar todos';
  btnValidarTodos.disabled = !pendentesDeValidar.length;
  btnValidarTodos.onclick = async ()=>{
    pendentesDeValidar.forEach(ped=> validarSleevePedido(ped, pedidosPorNumero[ped], coberturaPorCodigo));
    await saveState();
    renderTudoSleeve();
  };
}

// Modal "quais OPs vão fazer": lista as OPs pendentes (não validadas) de um
// código, aberta ao clicar no item na tela "Saldo de estoque".

  const overlay = document.getElementById('overlay');
  const modal = document.getElementById('modal');
  const ops = sleeveOPs.filter(o=>o.codigo===codigo && !o.validado && !o.saiuDaFila)
    .sort((a,b)=> new Date(a.termino||0) - new Date(b.termino||0));
  const nome = ops.length ? ops[0].nome : '';
  modal.innerHTML = `
    <div class="modal-head">
      <div><h2 style="font-size:16px;">${codigo}</h2><p>${nome}</p></div>
      <button class="close-x" id="modal-close">&times;</button>
    </div>
    <div class="hist-table-wrap">

// mostrado em "Saldo de estoque" é sempre a soma das quantidades daqui —

function itensSleeveConhecidos(){
  const mapa = {};
  sleeveItensManuais.forEach(it=>{ mapa[it.codigo] = it.nome; });
  sleeveOPs.forEach(o=>{ if(o.codigo) mapa[o.codigo] = o.nome; }); // pedido real sobrescreve o nome cadastrado manualmente

}
// Autocomplete próprio pra escolher item (código/nome) nos cards de sleeve —
// substitui o <datalist> nativo, que trunca o texto das opções sem dar pra
// controlar o tamanho; aqui a lista mostra o nome inteiro, sem cortar.
function setupItemAutocomplete(inputId, panelId){
  const input = document.getElementById(inputId);
  const panel = document.getElementById(panelId);
  function render(){
    const itensConhecidos = itensSleeveConhecidos();
    const termo = input.value.trim().toLowerCase();
    const codigos = Object.keys(itensConhecidos).sort();
    const filtrados = termo
      ? codigos.filter(cod=> cod.toLowerCase().includes(termo) || itensConhecidos[cod].toLowerCase().includes(termo))
      : codigos;
    panel.innerHTML = filtrados.length
      ? filtrados.slice(0,200).map(cod=>`<div class="item-autocomplete-option" data-codigo="${cod}"><b>${cod}</b> — ${itensConhecidos[cod]}</div>`).join('')
      : `<div class="item-autocomplete-empty">Nenhum item encontrado.</div>`;
    panel.style.display = 'block';
    panel.querySelectorAll('.item-autocomplete-option').forEach(opt=>{
      opt.addEventListener('mousedown', (e)=>{
        e.preventDefault();
        const cod = opt.dataset.codigo;
        input.value = `${cod} — ${itensConhecidos[cod]}`;
        input.dispatchEvent(new Event('input'));
        panel.style.display = 'none';
      });
    });
  }
  input.addEventListener('input', render);
  input.addEventListener('focus', render);
  input.addEventListener('blur', ()=>{ setTimeout(()=>{ panel.style.display = 'none'; }, 150); });
}

// Card "Estoque segregado": separa uma quantidade de um lote já recebido
// (o restante dos dados — gráfica, NF, fabricação, validade — vem herdado
// do recebimento daquele lote). Descontado do saldo em "Saldo de estoque".
function lotesDisponiveisDoItem(codigo){
  const porLote = {};
  sleeveRecebimentos.filter(r=>r.codigo===codigo).forEach(r=>{
    const key = r.lote || '(sem lote)';
    if(!porLote[key]) porLote[key] = { key, lote:r.lote||'', grafica:r.grafica, nf:r.nf, fabricacao:r.fabricacao, validade:r.validade, recebido:0 };
    porLote[key].recebido += (r.quantidade||0);
  });
  const segregadoPorLote = {};
  sleeveSegregados.filter(s=>s.codigo===codigo).forEach(s=>{
    const key = s.lote || '(sem lote)';
    segregadoPorLote[key] = (segregadoPorLote[key]||0) + (s.quantidade||0);
  });
  return Object.values(porLote).map(info=>{
    const segregado = segregadoPorLote[info.key] || 0;
    return { ...info, segregado, disponivel: info.recebido - segregado };
  });
}

    const fileInput = document.getElementById('sleeve-import-file');
    const resultDiv = document.getElementById('sleeve-import-result');
    const arquivo = fileInput.files[0];
    if(!arquivo){
      resultDiv.innerHTML = '<span style="color:var(--red);">Selecione o arquivo antes de processar.</span>';
      return;
    }
    resultDiv.innerHTML = '<span style="color:var(--text-dim);">Lendo arquivo...</span>';
    let texto;
    try{ texto = await arquivo.text(); }
    catch(e){ resultDiv.innerHTML = `<span style="color:var(--red);">Erro ao ler o arquivo: ${e.message}</span>`; return; }
    const novasOPs = parseSleeveRGER339(texto);
    if(!novasOPs.length){
      resultDiv.innerHTML = '<span style="color:var(--red);">Não encontrei nenhuma OP reconhecível nesse arquivo. Confira o formato do relatório.</span>';
      return;
    }
    mergeSleeveOPs(novasOPs);
    saveState();
    renderTudoSleeve();
    closeModal();
  });
